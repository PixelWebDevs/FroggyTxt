  function sanitizeRichText(html) {
    const parser = new DOMParser();
    const parsed = parser.parseFromString(
      '<div>' + String(html || '') + '</div>',
      'text/html'
    );

    const root = parsed.body.firstElementChild;

    const allowed = new Set([
      'DIV', 'P', 'BR', 'SPAN', 'B', 'STRONG', 'I', 'EM',
      'U', 'S', 'STRIKE', 'UL', 'OL', 'LI',
      'BLOCKQUOTE', 'SUB', 'SUP'
    ]);

    const blocked = new Set([
      'SCRIPT', 'STYLE', 'IFRAME', 'OBJECT', 'EMBED', 'SVG',
      'MATH', 'FORM', 'INPUT', 'BUTTON', 'VIDEO', 'AUDIO'
    ]);

    const safeStyle = sourceStyle => {
      const result = [];
      const color = sourceStyle.color;
      const background = sourceStyle.backgroundColor;
      const fontSize = sourceStyle.fontSize;
      const fontFamily = sourceStyle.fontFamily;
      const fontWeight = sourceStyle.fontWeight;
      const fontStyle = sourceStyle.fontStyle;
      const decoration = sourceStyle.textDecoration;
      const align = sourceStyle.textAlign;

      if (color && !/url|expression/i.test(color)) {
        result.push('color:' + color);
      }

      if (background && !/url|expression/i.test(background)) {
        result.push('background-color:' + background);
      }

      if (
        fontSize &&
        /^(?:[8-9]|[1-7][0-9])(?:\.\d+)?(?:px|pt)$/i.test(fontSize.trim())
      ) {
        result.push('font-size:' + fontSize);
      }

      if (fontFamily && !/url|expression|[<>]/i.test(fontFamily)) {
        result.push('font-family:' + fontFamily);
      }

      if (/^(normal|bold|[1-9]00)$/i.test(fontWeight)) {
        result.push('font-weight:' + fontWeight);
      }

      if (/^(normal|italic|oblique)$/i.test(fontStyle)) {
        result.push('font-style:' + fontStyle);
      }

      if (
        /^(none|underline|line-through|underline line-through|line-through underline)$/i.test(decoration)
      ) {
        result.push('text-decoration:' + decoration);
      }

      if (/^(left|right|center|justify)$/i.test(align)) {
        result.push('text-align:' + align);
      }

      return result.join(';');
    };

    function clean(node) {
      if (node.nodeType === Node.TEXT_NODE) {
        return parsed.createTextNode(node.nodeValue || '');
      }

      if (node.nodeType !== Node.ELEMENT_NODE) return null;

      const tag = node.tagName.toUpperCase();
      if (blocked.has(tag)) return null;

      const children = Array.from(node.childNodes)
        .map(clean)
        .filter(Boolean);

      if (!allowed.has(tag)) {
        const fragment = parsed.createDocumentFragment();
        children.forEach(child => fragment.appendChild(child));
        return fragment;
      }

      const element = parsed.createElement(tag.toLowerCase());

      if (node.hasAttribute('style')) {
        const cleanCss = safeStyle(node.style);
        if (cleanCss) element.setAttribute('style', cleanCss);
      }

      children.forEach(child => element.appendChild(child));
      return element;
    }

    const fragment = parsed.createDocumentFragment();

    Array.from(root.childNodes)
      .map(clean)
      .filter(Boolean)
      .forEach(node => fragment.appendChild(node));

    const holder = parsed.createElement('div');
    holder.appendChild(fragment);

    return holder.innerHTML;
  }

  function escapeHTML(value) {

  function selectionNodePath(node) {
    const path = [];

    while (node && node !== editor) {
      const parent = node.parentNode;
      if (!parent) return null;

      path.unshift(
        Array.prototype.indexOf.call(parent.childNodes, node)
      );

      node = parent;
    }

    return node === editor ? path : null;
  }

  function snapshotEditorSelection() {
    const selection = window.getSelection();
    if (!selection || !selection.rangeCount) return null;

    const range = selection.getRangeAt(0);
    const start = selectionNodePath(range.startContainer);
    const end = selectionNodePath(range.endContainer);

    if (!start || !end) return null;

    return {
      start,
      startOffset: range.startOffset,
      end,
      endOffset: range.endOffset
    };
  }

  function restoreHistorySelection(position) {
    editor.focus({ preventScroll: true });

    const selection = window.getSelection();
    if (!selection) return;

    const atPath = path => {
      let node = editor;

      for (const index of path) {
        if (!node.childNodes[index]) return null;
        node = node.childNodes[index];
      }

      return node;
    };

    const range = document.createRange();

    try {
      const start = position && atPath(position.start);
      const end = position && atPath(position.end);

      if (!start || !end) {
        range.selectNodeContents(editor);
        range.collapse(false);
      } else {
        range.setStart(
          start,
          Math.min(
            position.startOffset,
            start.nodeType === Node.TEXT_NODE
              ? start.length
              : start.childNodes.length
          )
        );

        range.setEnd(
          end,
          Math.min(
            position.endOffset,
            end.nodeType === Node.TEXT_NODE
              ? end.length
              : end.childNodes.length
          )
        );
      }
    } catch (error) {
      range.selectNodeContents(editor);
      range.collapse(false);
    }

    selection.removeAllRanges();
    selection.addRange(range);
    savedRange = range.cloneRange();
  }

  function updateHistoryButtons() {
    const history = noteHistories.get(currentNoteId);

    $('undoButton').disabled =
      !history || history.index === 0;

    $('redoButton').disabled =
      !history || history.index === history.steps.length - 1;
  }

  function recordEditorHistory() {
    let history = noteHistories.get(currentNoteId);

    if (!history) {
      history = {
        steps: [{
          html: currentNote().content,
          selection: null
        }],
        index: 0
      };

      noteHistories.set(currentNoteId, history);
    }

    const html = editor.innerHTML;
    const position = snapshotEditorSelection();

    if (history.steps[history.index].html === html) {
      if (position) {
        history.steps[history.index].selection = position;
      }
    } else {
      history.steps.splice(history.index + 1);
      history.steps.push({ html, selection: position });

      if (history.steps.length > MAX_HISTORY_ENTRIES) {
        history.steps.shift();
      }

      history.index = history.steps.length - 1;
    }

    updateHistoryButtons();
  }

  function stepThroughHistory(direction) {
    const history = noteHistories.get(currentNoteId);
    if (!history) return;

    const nextIndex = history.index + direction;

    if (nextIndex < 0 || nextIndex >= history.steps.length) return;

    history.index = nextIndex;
    const step = history.steps[nextIndex];

    editor.innerHTML = step.html;

    const note = currentNote();
    note.content = step.html;
    note.updatedAt = Date.now();

    restoreHistorySelection(step.selection);
    saveData();
    refreshHeader();
    updateHistoryButtons();
  }

  function loadCurrentNote() {
    const note = currentNote();

    editor.innerHTML = sanitizeRichText(note.content);
    note.content = editor.innerHTML;
    savedRange = null;

    editor.style.fontSize = data.settings.defaultTextSize + 'px';
    editor.style.fontFamily =
      data.settings.textFont || DEFAULTS.defaultFont;

    const history = noteHistories.get(currentNoteId);

    if (
      !history ||
      history.steps[history.index].html !== editor.innerHTML
    ) {
      noteHistories.set(currentNoteId, {
        steps: [{
          html: editor.innerHTML,
          selection: null
        }],
        index: 0
      });
    }

    updateHistoryButtons();
    refreshHeader();
    saveData();
  }

  function saveCurrentNote() {
    const note = currentNote();

    recordEditorHistory();
    note.content = editor.innerHTML;
    note.updatedAt = Date.now();

    saveData();
    refreshHeader();
  }

  function captureSelection() {
    const selection = window.getSelection();
    if (!selection || !selection.rangeCount) return;

    const range = selection.getRangeAt(0);

    if (
      editor.contains(range.startContainer) &&
      editor.contains(range.endContainer)
    ) {
      savedRange = range.cloneRange();
    }
  }

  function restoreSelection() {
    const selection = window.getSelection();
    if (!selection) return false;

    const valid =
      savedRange &&
      editor.contains(savedRange.startContainer) &&
      editor.contains(savedRange.endContainer);

    const range = valid
      ? savedRange.cloneRange()
      : document.createRange();

    if (!valid) {
      range.selectNodeContents(editor);
      range.collapse(false);
    }

    selection.removeAllRanges();
    selection.addRange(range);
    savedRange = range.cloneRange();

    return true;
  }

function withTextSelection(action) {
  const control = document.activeElement;

  applyingTextStyle = true;
  editor.setAttribute('contenteditable', 'true');

  try {
    const selection = window.getSelection();
    if (!selection) return;

    // Apply formatting to the entire note.
    const range = document.createRange();
    range.selectNodeContents(editor);

    selection.removeAllRanges();
    selection.addRange(range);
    savedRange = range.cloneRange();

    action();
    captureSelection();
    syncTextStyleControls();
  } finally {
    // Leave the cursor at the end so typing does not replace the note.
    const selection = window.getSelection();

    if (
      selection &&
      selection.rangeCount &&
      editor.contains(selection.getRangeAt(0).startContainer) &&
      editor.contains(selection.getRangeAt(0).endContainer)
    ) {
      const cursor = selection.getRangeAt(0).cloneRange();
      cursor.collapse(false);

      selection.removeAllRanges();
      selection.addRange(cursor);
      savedRange = cursor.cloneRange();
    }

    applyingTextStyle = false;

    if (!$('textOverlay').hidden) {
      editor.blur();
      editor.setAttribute('contenteditable', 'false');

      if (control && $('textOverlay').contains(control)) {
        control.focus({ preventScroll: true });
      }
    }
  }
}

  function syncTextStyleControls() {
    document.querySelectorAll('[data-command]').forEach(button => {
      const active = document.queryCommandState(
        button.dataset.command
      );

      button.classList.toggle('active', active);
      button.setAttribute('aria-pressed', String(active));
    });

    if (!savedRange) return;

    let node = savedRange.startContainer;

    if (node.nodeType === Node.ELEMENT_NODE) {
      node = node.childNodes[savedRange.startOffset] || node;
    }

    while (
      node.nodeType === Node.ELEMENT_NODE &&
      node.firstChild
    ) {
      node = node.firstChild;
    }

    const element = node.nodeType === Node.TEXT_NODE
      ? node.parentElement
      : node;

    const style = getComputedStyle(element);
    const size = Math.round(parseFloat(style.fontSize)) || 22;

    $('selectionSize').value = size;
    $('selectionSizeValue').textContent = size + ' px';

    const hex = color => {
      const parts = color.match(
        /^rgba?\((\d+),\s*(\d+),\s*(\d+)/
      );

      return parts
        ? '#' + parts.slice(1, 4).map(
            n => Number(n).toString(16).padStart(2, '0')
          ).join('')
        : null;
    };

    const color = hex(style.color);
    if (color) $('selectionColor').value = color;
  }

  function setInlineStyle(property, value) {
    const selection = window.getSelection();
    const range = selection.getRangeAt(0);

    if (!range.collapsed && property === 'backgroundColor') {
      document.execCommand('hiliteColor', false, value);
      captureSelection();
      return;
    }

    if (!range.collapsed) {
      let ancestor = range.commonAncestorContainer;

      if (ancestor.nodeType === Node.TEXT_NODE) {
        ancestor = ancestor.parentElement;
      }

      while (
        ancestor &&
        ancestor !== editor &&
        ancestor.tagName === 'SPAN' &&
        range.toString() === ancestor.textContent
      ) {
        const parent = ancestor.parentElement;
        range.selectNode(ancestor);
        ancestor = parent;
      }
    }

    const span = document.createElement('span');
    span.style[property] = value;

    const next = document.createRange();

    if (range.collapsed) {
      const marker = document.createTextNode('\u200B');
      span.appendChild(marker);
      range.insertNode(span);

      next.setStart(marker, 1);
      next.collapse(true);
    } else {
      const content = range.extractContents();

      content.querySelectorAll('[style]').forEach(element => {
        element.style[property] = '';

        if (!element.getAttribute('style')) {
          element.removeAttribute('style');
        }
      });

      span.appendChild(content);
      range.insertNode(span);
      next.selectNodeContents(span);
    }

    selection.removeAllRanges();
    selection.addRange(next);
    savedRange = next.cloneRange();
  }

  function applyInlineStyle(property, value) {
    withTextSelection(() => {
      setInlineStyle(property, value);
      saveCurrentNote();
    });
  }

  function applyCommand(command) {
    withTextSelection(() => {
      const range = window.getSelection().getRangeAt(0);

      if (
        range.collapsed &&
        ['bold', 'italic', 'underline', 'strikeThrough'].includes(command)
      ) {
        const active = document.queryCommandState(command);

        if (command === 'bold') {
          setInlineStyle('fontWeight', active ? '400' : '700');
        } else if (command === 'italic') {
          setInlineStyle('fontStyle', active ? 'normal' : 'italic');
        } else {
          let underline = document.queryCommandState('underline');
          let strike = document.queryCommandState('strikeThrough');

          if (command === 'underline') {
            underline = !active;
          } else {
            strike = !active;
          }

          const decoration = [
            underline ? 'underline' : '',
            strike ? 'line-through' : ''
          ].filter(Boolean).join(' ') || 'none';

          setInlineStyle('textDecoration', decoration);
        }
      } else {
        document.execCommand(command, false, null);
      }

      captureSelection();
      saveCurrentNote();
    });
  }

  function cleanEmptyMarkers() {
    editor.querySelectorAll('span').forEach(span => {
      if (span.textContent === '\u200B') {
        span.textContent = '';
      }
    });
  }

  function speakText() {
    if (!('speechSynthesis' in window)) {
      showToast('Speech playback is not available in this browser');
      return;
    }

    if (speaking) {
      window.speechSynthesis.cancel();
      speaking = false;
      $('speakLabel').textContent = 'Speak';
      return;
    }

    const selection = window.getSelection();
    let text = '';

if (data.settings.selectedSpeakOnly) {
  const range =
    selection && selection.rangeCount
      ? selection.getRangeAt(0)
      : null;

  const validSelection =
    range &&
    !range.collapsed &&
    editor.contains(range.startContainer) &&
    editor.contains(range.endContainer);

  if (!validSelection) {
    showToast('Select text in your note before pressing Speak');
    return;
  }

  text = range.toString().replace(/\u200B/g, '').trim();

  if (!text) {
    showToast('Select text in your note before pressing Speak');
    return;
  }
} else {
  text = editor.innerText.replace(/\u200B/g, '').trim();
}

    if (!text) {
      showToast('Type something before speaking');
      return;
    }

    window.speechSynthesis.cancel();

    const utterance = new SpeechSynthesisUtterance(text);
    utterance.rate = Number(data.settings.speechRate) || 1;
    utterance.pitch = Number(data.settings.speechPitch) || 1;

    const voice = window.speechSynthesis.getVoices().find(
      v => v.name === data.settings.speechVoice
    );

    if (voice) utterance.voice = voice;

    utterance.onstart = () => {
      speaking = true;
      $('speakLabel').textContent = 'Stop';
    };

    utterance.onend = utterance.onerror = () => {
      speaking = false;
      $('speakLabel').textContent = 'Speak';
    };

    window.speechSynthesis.speak(utterance);
  }

  function populateVoices() {
    const select = $('speechVoice');
    if (!('speechSynthesis' in window)) return;

    const voices = window.speechSynthesis.getVoices();
    const selected = data.settings.speechVoice;

    select.innerHTML = '<option value="">System default</option>';

    voices.forEach(voice => {
      const option = document.createElement('option');
      option.value = voice.name;
      option.textContent =
        voice.name + (voice.lang ? ' · ' + voice.lang : '');

      select.appendChild(option);
    });

    select.value = selected;
  }

  function normalizeDeletedNotes(notes) {
    return (Array.isArray(notes) ? notes : [])
      .filter(note => note && typeof note === 'object')
      .map(note => ({
        id: String(note.id || uid()),
        content: sanitizeRichText(
          typeof note.content === 'string' ? note.content : ''
        ),
        folderId: note.folderId || null,
        updatedAt: Number(note.updatedAt) || Date.now(),
        deletedAt: Number(note.deletedAt) || Date.now()
      }));
  }

        remove.type = 'button';
        remove.className = 'small-action danger-action';
        remove.textContent = 'Delete permanently';

        remove.addEventListener('click', () => {

  function recoverDeletedNote(id) {
    const note = data.deletedNotes.find(note => note.id === id);
    if (!note) return;

    const restored = {
      id: data.notes.some(active => active.id === note.id)
        ? uid()
        : note.id,
      content: sanitizeRichText(note.content),
      folderId: data.folders.some(folder => folder.id === note.folderId)
        ? note.folderId
        : null,
      updatedAt: Date.now()
    };

    data.notes.push(restored);
    data.deletedNotes = data.deletedNotes.filter(note => note.id !== id);

    saveData();
    renderLibrary();
    showToast('Note recovered');
  }

  function permanentlyDeleteNote(id) {
    const note = data.deletedNotes.find(note => note.id === id);

    if (!note || $('permanentDeleteConfirmation')) return;

    const overlay = document.createElement('div');
    overlay.id = 'permanentDeleteConfirmation';
    overlay.className = 'overlay confirm-overlay';

    overlay.innerHTML = `
      <section
        class="sheet confirm-sheet"
        role="alertdialog"
        aria-modal="true"
        aria-labelledby="permanentDeleteTitle"
        aria-describedby="permanentDeleteMessage"
      >
        <div class="confirm-title" id="permanentDeleteTitle">
          Delete permanently?
        </div>
        <div class="confirm-message" id="permanentDeleteMessage"></div>
        <div class="confirm-actions">
          <button type="button" class="small-action" id="cancelPermanentDelete">
            Cancel
          </button>
          <button type="button" class="small-action danger-action" id="confirmPermanentDelete">
            Delete
          </button>
        </div>
      </section>
    `;

    overlay.querySelector('#permanentDeleteMessage').textContent =
      'Permanently delete “' + titleFor(note) +
      '”? This cannot be undone.';

    const previousFocus = document.activeElement;

    function closeConfirmation() {
      document.removeEventListener('keydown', handleKey, true);
      overlay.remove();

      if (previousFocus && previousFocus.isConnected) {
        previousFocus.focus({ preventScroll: true });
      }
    }

    function handleKey(event) {
      if (event.key === 'Escape') {
        event.preventDefault();
        event.stopImmediatePropagation();
        closeConfirmation();
      } else if (event.key === 'Tab') {
        event.preventDefault();

        const cancel = overlay.querySelector('#cancelPermanentDelete');
        const confirm = overlay.querySelector('#confirmPermanentDelete');

        (document.activeElement === cancel ? confirm : cancel).focus();
      }
    }

    overlay.querySelector('#cancelPermanentDelete')
      .addEventListener('click', closeConfirmation);

    overlay.querySelector('#confirmPermanentDelete')
      .addEventListener('click', () => {
        data.deletedNotes = data.deletedNotes.filter(
          note => note.id !== id
        );

        saveData();
        closeConfirmation();
        renderLibrary();
        showToast('Note permanently deleted');
      });

    overlay.addEventListener('click', event => {
      if (event.target === overlay) closeConfirmation();
    });

    document.addEventListener('keydown', handleKey, true);
    document.body.appendChild(overlay);

    overlay.querySelector('#cancelPermanentDelete').focus();
  }

  function renderLibrary() {
    const body = $('libraryBody');
    body.innerHTML = '';

    if (activeFolderId) {
      renderFolderNotes(body, activeFolderId);
      return;
    }

    const segmented = document.createElement('div');
    segmented.className = 'segmented';

    segmented.innerHTML =
      '<button class="segment ' +
      (libraryTab === 'recent' ? 'active' : '') +
      '" data-library-tab="recent">Recent</button>' +
      '<button class="segment ' +
      (libraryTab === 'folders' ? 'active' : '') +
      '" data-library-tab="folders">Folders</button>' +
      '<button class="segment ' +
      (libraryTab === 'deleted' ? 'active' : '') +
      '" data-library-tab="deleted">Deleted</button>';

    body.appendChild(segmented);

    segmented.querySelectorAll('[data-library-tab]').forEach(button => {
      button.addEventListener('click', () => {
        libraryTab = button.dataset.libraryTab;
        folderFormMode = '';
        editingFolderId = null;
        renderLibrary();
      });
    });

    if (libraryTab === 'recent') {
      renderRecent(body);
    } else if (libraryTab === 'deleted') {
      renderDeleted(body);
    } else {
      renderFolders(body);
    }
  }

  function openFolderDeleteConfirmation(folderId) {
    const folder = data.folders.find(f => f.id === folderId);
    if (!folder) return;

    pendingFolderDeleteId = folder.id;

    $('folderDeleteMessage').textContent =
      'Delete Folder “' + folder.name +
      '”?\nThe notes will be transferred to Recent.';

    $('folderDeleteOverlay').hidden = false;
  }

  function closeFolderDeleteConfirmation() {
    $('folderDeleteOverlay').hidden = true;
    pendingFolderDeleteId = null;
  }

  function confirmFolderDeletion() {
    const folder = data.folders.find(
      f => f.id === pendingFolderDeleteId
    );

    if (!folder) {
      closeFolderDeleteConfirmation();
      return;
    }

    const now = Date.now();

    data.notes.forEach(note => {
      if (note.folderId === folder.id) {
        note.folderId = null;
        note.updatedAt = now;
      }
    });

    data.folders = data.folders.filter(f => f.id !== folder.id);

    if (activeFolderId === folder.id) activeFolderId = null;

    folderFormMode = '';
    editingFolderId = null;

    closeFolderDeleteConfirmation();
    saveData();
    refreshHeader();
    renderLibrary();
    showToast('Folder deleted. Notes moved to Recent.');
  }

  function renderFolderNotes(body, folderId) {
    const folder = data.folders.find(f => f.id === folderId);

    if (!folder) {
      activeFolderId = null;
      renderLibrary();
      return;
    }

    const back = document.createElement('div');
    back.className = 'back-row';

    const backButton = document.createElement('button');
    backButton.innerHTML = '‹ &nbsp; Folders';

    backButton.addEventListener('click', () => {
      activeFolderId = null;
      libraryTab = 'folders';
      renderLibrary();
    });

    back.appendChild(backButton);
    body.appendChild(back);

    const label = document.createElement('div');
    label.className = 'section-label';
    label.textContent = folder.name;
    body.appendChild(label);

    const list = document.createElement('div');
    list.className = 'note-list';

    data.notes
      .filter(n => n.folderId === folderId)
      .sort((a, b) => b.updatedAt - a.updatedAt)
      .forEach(note => {
        const row = document.createElement('div');
        row.className = 'note-row';

        const main = document.createElement('button');
        main.className = 'note-main';

        main.innerHTML =
          '<div class="note-name">' +
          escapeHTML(titleFor(note)) +
          '</div><div class="note-preview">' +
          escapeHTML(previewFor(note)) +
          '</div>';

        main.addEventListener('click', () => switchNote(note.id));

        const move = document.createElement('button');
        move.className = 'mini-button';
        move.title = 'Move note';
        move.setAttribute('aria-label', 'Move note');

        move.innerHTML =
          '<svg viewBox="0 0 24 24"><path d="M4 7h16M4 12h10M4 17h16M16 9l4 3-4 3"/></svg>';

        move.addEventListener('click', () => {
          renderMoveMenu(body, note.id);
        });

        const remove = document.createElement('button');
        remove.className = 'mini-button';
        remove.title = 'Delete note';
        remove.setAttribute('aria-label', 'Delete note');

        remove.innerHTML =
          '<svg viewBox="0 0 24 24"><path d="M4 7h16M9 7V4h6v3M7 7l1 13h8l1-13M10 11v5M14 11v5"/></svg>';

        remove.addEventListener('click', () => deleteNote(note.id));

        row.append(main, move, remove);
        list.appendChild(row);
      });

    if (!list.children.length) {
      list.innerHTML =
        '<div class="empty-state">This folder is empty. Move a note here from Recent.</div>';
    }

    body.appendChild(list);
  }

  function renderMoveMenu(body, noteId) {
    const old = body.querySelector('.move-picker');
    if (old) old.remove();

    const picker = document.createElement('div');
    picker.className = 'inline-form move-picker';

    const label = document.createElement('div');
    label.className = 'control-name';
    label.textContent = 'Move note to';
    picker.appendChild(label);

    const options = [
      { id: null, name: 'Recent', color: '#92909a' },
      ...data.folders
    ];

    options.forEach(folder => {
      const button = document.createElement('button');
      button.className = 'small-action';
      button.style.textAlign = 'left';
      button.style.display = 'flex';
      button.style.alignItems = 'center';
      button.style.gap = '9px';

      button.innerHTML =
        '<span class="folder-dot" style="background:' +
        folder.color +
        '"></span>' +
        escapeHTML(folder.name);

      button.addEventListener('click', () => {
        const note = data.notes.find(n => n.id === noteId);

        if (note) {
          note.folderId = folder.id;
          note.updatedAt = Date.now();

          saveData();
          refreshHeader();
          renderLibrary();
          showToast('Note moved to ' + folder.name);
        }
      });

      picker.appendChild(button);
    });

    const cancel = document.createElement('button');
    cancel.className = 'small-action';
    cancel.textContent = 'Cancel';

    cancel.addEventListener('click', () => picker.remove());
    picker.appendChild(cancel);

    body.appendChild(picker);

    picker.scrollIntoView({
      block: 'nearest',
      behavior: 'smooth'
    });
  }

  function deleteNote(id) {
    const noteToDelete = data.notes.find(note => note.id === id);
    if (!noteToDelete) return;

    const deletingCurrentNote = currentNoteId === id;

    if (deletingCurrentNote) saveCurrentNote();

    data.deletedNotes.push({
      ...noteToDelete,
      deletedAt: Date.now()
    });

    data.notes = data.notes.filter(note => note.id !== id);
    noteHistories.delete(id);

    if (data.notes.length === 0) {
      const replacementNote = {
        id: uid(),
        content: '',
        folderId: null,
        updatedAt: Date.now()
      };

      data.notes.push(replacementNote);
      currentNoteId = replacementNote.id;
    } else if (deletingCurrentNote) {
      const nextNote = [...data.notes]
        .sort((a, b) => b.updatedAt - a.updatedAt)[0];

      currentNoteId = nextNote.id;
    }

    if (deletingCurrentNote) loadCurrentNote();

    saveData();
    renderLibrary();
    showToast('Note moved to Deleted');
  }

  $('importFile').addEventListener('change', event => {
    const file = event.target.files && event.target.files[0];
    if (!file) return;

    const reader = new FileReader();

    reader.onload = () => {
      try {
        const imported = JSON.parse(String(reader.result));

        if (
          !imported ||
          !Array.isArray(imported.notes) ||
          !Array.isArray(imported.folders)
        ) {
          throw new Error('Invalid backup');
        }

        if (!window.confirm(
          'Replace the notes and settings on this device with this backup?'
        )) return;

        data = imported;

        data.settings = {
          ...DEFAULTS,
          ...(data.settings || {})
        };

        data.deletedNotes = normalizeDeletedNotes(
          data.deletedNotes
        );

        noteHistories.clear();

        data.notes = data.notes.map(n => ({
          id: String(n.id || uid()),
          content: sanitizeRichText(n.content),
          folderId: n.folderId || null,
          updatedAt: Number(n.updatedAt) || Date.now()
        }));

        data.folders = data.folders.map(f => ({
          id: String(f.id || uid()),
          name: String(f.name || 'Folder'),
          color: /^#[0-9a-f]{6}$/i.test(f.color)
            ? f.color
            : '#8b82db'
        }));

        if (!data.notes.length) {
          data.notes.push({
            id: uid(),
            content: '',
            folderId: null,
            updatedAt: Date.now()
          });
        }

        currentNoteId = data.notes.some(
          n => n.id === data.currentNoteId
        )
          ? data.currentNoteId
          : data.notes[0].id;

        saveData();
        applySettings();
        loadCurrentNote();
        showToast('Backup imported');
        closeSheet('settingsOverlay');
      } catch (error) {
        showToast('That backup file could not be opened');
      }

      event.target.value = '';
    };

    reader.onerror = () => {
      showToast('Could not read that file');
    };

    reader.readAsText(file);
  });
